21	    stop: int,

24	) -> None:
25	    """Write Wilder's RSI for the gap-free run ``close[start:stop]`` into ``result``."""
26	    n = stop - start
27	
28	    if period < 1 or period >= n:
29	        return
30	
31	    # Calculate initial sums for the first period
32	    sum_gain = 0.0
33	    sum_loss = 0.0
34	
35	    for i in range(start + 1, start + period + 1):
36	        change = close[i] - close[i - 1]
37	        if change > 0:
38	            sum_gain += change
39	        else:
40	            sum_loss += -change
41

43	    avg_gain = sum_gain / period
44	    avg_loss = sum_loss / period

46	    if avg_loss != 0:

48	        result[start + period] = 100.0 - (100.0 / (1.0 + rs))
49	    # When avg_loss is 0, TA-Lib returns 0 if avg_gain is also 0

52	    else:

56	    # Pre-compute constant

58	    inv_period = 1.0 / period
59
